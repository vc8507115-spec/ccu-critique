      const navLabel = document.getElementById('currentNavLabel');
      const membersTabsContainer = document.getElementById('groupMembersTabsContainer');
      const groupSectionCard = document.getElementById('groupSectionCard');
      const syncAllBtn = document.getElementById('syncAllBtn');
      const idTagLabel = document.getElementById('idTagLabel');
      const activeMemberSpot = document.getElementById('activeMemberSpot');

      if (sessionData.isGroupMode) {
        if (idTagLabel) idTagLabel.innerText = 'GROUP';
        if (navLabel) navLabel.innerText = `第 ${currentItem.groupName} 組 (${index + 1}/${sessionData.items.length})`;
        document.getElementById('studentIdDisplay').innerText = `G${index + 1}`;
        document.getElementById('studentNameDisplay').innerText = currentItem.groupName;
        document.getElementById('advisorDisplay').innerText = currentItem.advisor;

        if (activeMemberSpot) activeMemberSpot.classList.remove('hidden');
        if (membersTabsContainer) membersTabsContainer.classList.remove('hidden');
        if (groupSectionCard) groupSectionCard.classList.remove('hidden');
        if (syncAllBtn) syncAllBtn.classList.remove('hidden');
      } else {
        if (idTagLabel) idTagLabel.innerText = 'ID';
        if (navLabel) navLabel.innerText = `${currentItem.id} ${currentItem.name} (${index + 1}/${sessionData.items.length})`;
        document.getElementById('studentIdDisplay').innerText = currentItem.id;
        document.getElementById('studentNameDisplay').innerText = currentItem.name;
        document.getElementById('advisorDisplay').innerText = currentItem.advisor;

        if (activeMemberSpot) {
          activeMemberSpot.classList.add('hidden');
          activeMemberSpot.innerHTML = '';
        }
        if (membersTabsContainer) membersTabsContainer.classList.add('hidden');
        if (groupSectionCard) groupSectionCard.classList.add('hidden');
        if (syncAllBtn) syncAllBtn.classList.add('hidden');
      }

      const mem = reviewMemory[index];
      if (mem) {
        formState.memberNotes = mem.memberNotes ? { ...mem.memberNotes } : {};
        formState.savedMembers = mem.savedMembers ? { ...mem.savedMembers } : {};
        formState.currentPhoto = mem.photo || null;
        document.getElementById('savedIndicator').classList.remove('hidden');
      } else {
        formState.memberNotes = {};
        formState.savedMembers = {};
        formState.currentPhoto = null;
        document.getElementById('savedIndicator').classList.add('hidden');
      }

      const photoContainer = document.getElementById('photoContainer');
      if (photoContainer) {
        photoContainer.innerHTML = renderPhotoSectionHTML();
      }

      if (sessionData.isGroupMode) {
        formState.groupScores = {};
        formState.indivScores = {};

        (sessionData.groupCriteria || []).forEach((_, gi) => {
          formState.groupScores[gi] = (mem && mem.groupScores && mem.groupScores[gi] !== undefined)
            ? mem.groupScores[gi]
            : null;
        });

        if (currentItem.members) {
          currentItem.members.forEach(m => {
            formState.indivScores[m.id] = {};
            (sessionData.indivCriteria || []).forEach((_, ii) => {
              const savedVal = mem && mem.indivScores && mem.indivScores[m.id] && mem.indivScores[m.id][ii];

          });

        (sessionData.criteria || []).forEach((_, i) => {

        document.getElementById('notes').value = currentNotes;
      }

      calcAndRefreshTotals();

    }

    function selectMember(idx) {
      const currentItem = sessionData.items[currentIndex];
      if (!currentItem || !currentItem.members || !currentItem.members[idx]) return;

      const oldMember = currentItem.members[selectedMemberIndex];
      const notesInput = document.getElementById('notes');
      if (oldMember && notesInput) {
        formState.memberNotes[oldMember.id] = notesInput.value;
      }

      selectedMemberIndex = idx;
      const newMember = currentItem.members[idx];

      const hasSaved = formState.savedMembers && formState.savedMembers[newMember.id] !== undefined;
      if (!hasSaved) {
        if (!formState.indivScores) formState.indivScores = {};
        if (!formState.indivScores[newMember.id]) {
          formState.indivScores[newMember.id] = {};
        }
        if (notesInput) notesInput.value = formState.memberNotes[newMember.id] || '';
      } else {
        if (notesInput) {
          notesInput.value = formState.memberNotes[newMember.id] || '';
        }
      }

      renderMemberPillTabs();
      renderCriteriaSections();
      calcAndRefreshTotals();
      setButtonLoading(false);
    }

    function setButtonLoading(loading) {
      const btn = document.getElementById('submitBtn');
      if (!btn) return;

      if (loading) {
        btn.disabled = true;
        btn.classList.add('opacity-80', 'cursor-not-allowed');
        btn.innerHTML = `
          <span class="inline-block w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin mr-2"></span>
          <span class="tracking-wide">儲存中...</span>
        `;
      } else {
        btn.disabled = false;
        btn.classList.remove('opacity-80', 'cursor-not-allowed');
        updateSubmitButtonLabel();
      }
    }

    function updateSubmitButtonLabel() {

      if (!btn || !sessionData) return;

      btn.classList.remove('opacity-80', 'cursor-not-allowed');

      let labelText = '儲存本筆評分並前往下一位';

      }

        <span id="submitBtnText" class="tracking-wide drop-shadow-[0_1px_2px_rgba(0,0,0,0.7)]">${labelText}</span>
        <span class="group-hover:translate-x-1.5 transition-transform text-lg font-times">➔</span>
      `;
    }

    function syncCurrentMemberScoresToAll() {
      const currentItem = sessionData.items[currentIndex];
      if (!currentItem || !currentItem.members) return;
      const srcMember = currentItem.members[selectedMemberIndex];
      const srcScores = (formState.indivScores && formState.indivScores[srcMember.id]) || {};
      
      let hasScores = false;
      (sessionData.indivCriteria || []).forEach((_, ii) => {
        if (srcScores[ii] !== undefined && srcScores[ii] !== null && srcScores[ii] !== '') {
          hasScores = true;
        }
      });
      if (!hasScores) {
        showToast(`【${srcMember.name}】尚未輸入個人評分，無法同步！`, '⚠️');
        return;
      }

      const srcNotes = document.getElementById('notes').value.trim();

      currentItem.members.forEach(m => {
        formState.indivScores[m.id] = { ...srcScores };
        if (srcNotes) {
          formState.memberNotes[m.id] = srcNotes;
        }
      });
      showToast(`已將【${srcMember.name}】的分數帶給全組組員！`, '⚡');
      renderMemberPillTabs();
      calcAndRefreshTotals();
    }

      const row = document.createElement('div');
      row.className = 'p-3 rounded-xl transition-all flex justify-between items-center bg-[#10141a] border border-[#242b37] hover:border-[#ca6b2d]/60 shadow-sm';
      const displayVal = (val !== null && val !== undefined && val !== '') ? val : '';

      row.innerHTML = `
        <div class="flex items-center gap-3 pr-2 min-w-0">
          <div class="text-xs sm:text-sm font-bold text-[#f0f6fc] truncate">${name}</div>

        </div>
        <div class="flex items-center gap-2 shrink-0" onclick="event.stopPropagation()">
          <input type="number" inputmode="decimal" min="0" max="100" value="${displayVal}" placeholder="0"
            onfocus="handleCriterionInputFocus(this)"
            oninput="handleCriterionInput(${idx}, this.value, '${mode}', '${memberId || ''}')"
            onblur="handleCriterionInputBlur(${idx}, this, '${mode}', '${memberId || ''}')"
            class="w-16 h-10 text-center font-times font-black text-lg text-[#ffb573] placeholder:text-[#475263] metallic-engraved-input rounded-xl focus:outline-none focus:border-[#ff9f43] focus:ring-1 focus:ring-[#ff9f43]/50 transition">

    }

    function handleCriterionInputFocus(inputEl) {
      setTimeout(() => {
        try { inputEl.setSelectionRange(0, inputEl.value.length); } catch (e) { inputEl.select(); }
      }, 25);
    }

    function handleCriterionInput(idx, valueStr, mode, memberId) {
      const trimmed = String(valueStr).trim();
      let num = trimmed === '' ? null : parseFloat(trimmed);
      if (num !== null) {
        if (isNaN(num)) num = 0;
        if (num > 100) num = 100;
        if (num < 0) num = 0;
      }

      if (mode === 'group') {
        formState.groupScores[idx] = num;
      } else if (mode === 'indiv_member') {
        if (!formState.indivScores[memberId]) formState.indivScores[memberId] = {};
        formState.indivScores[memberId][idx] = num;
      } else {
        formState.scores[idx] = num;
      }
      calcAndRefreshTotals();
    }

    function handleCriterionInputBlur(idx, inputEl, mode, memberId) {
      let val = '';
      if (mode === 'group') {
        val = (formState.groupScores[idx] !== undefined && formState.groupScores[idx] !== null)
          ? formState.groupScores[idx]
          : '';
      } else if (mode === 'indiv_member') {
        val = (formState.indivScores[memberId] && formState.indivScores[memberId][idx] !== undefined && formState.indivScores[memberId][idx] !== null)
          ? formState.indivScores[memberId][idx] : '';
      } else {
        val = (formState.scores[idx] !== undefined && formState.scores[idx] !== null) ? formState.scores[idx] : '';
      }
      inputEl.value = val;
      calcAndRefreshTotals();
    }

    function calcAndRefreshTotals() {
      let groupSubtotalScore = 0;
      let activeMemberIndivSubtotal = 0;
      let combinedFinalScore = 0;

      const mainGaugeTitle = document.getElementById('mainGaugeTitle');
      const mainGaugeVal = document.getElementById('mainGaugeVal');
      const mainGaugeBreakdown = document.getElementById('mainGaugeBreakdown');
      const groupSubtotalDisplay = document.getElementById('groupSubtotalDisplay');
      const indivSubtotalDisplay = document.getElementById('indivSubtotalDisplay');

      if (sessionData.isGroupMode) {
        const currentItem = sessionData.items[currentIndex];
        const gRatio = (sessionData.groupRatio !== undefined && sessionData.groupRatio > 0) ? sessionData.groupRatio : 30;
        const iRatio = (sessionData.indivRatio !== undefined && sessionData.indivRatio > 0) ? sessionData.indivRatio : 40;
        const totalWeightScale = gRatio + iRatio;
        
        let gWeightedSum = 0;
        let gWeightTotal = 0;
        let allGroupEntered = true;
        const groupCriteriaList = sessionData.groupCriteria || [];
        if (groupCriteriaList.length === 0) {
          allGroupEntered = false;
        } else {
          for (let gi = 0; gi < groupCriteriaList.length; gi++) {
            const sc = formState.groupScores ? formState.groupScores[gi] : null;
            if (sc === undefined || sc === null || sc === '') {
              allGroupEntered = false;
              break;
            }
          }
        }

        if (allGroupEntered) {
          groupCriteriaList.forEach((crit, gi) => {
            const s = Number(formState.groupScores[gi]) || 0;
            gWeightedSum += s * crit.weight;
            gWeightTotal += crit.weight;
          });
          groupSubtotalScore = gWeightTotal > 0 ? (gWeightedSum / gWeightTotal) : 0;
        } else {
          groupSubtotalScore = 0;
        }

        const gRatioBadge = document.getElementById('groupRatioBadge');
        if (gRatioBadge) gRatioBadge.innerText = `佔比 ${gRatio}%`;
        const iRatioBadge = document.getElementById('indivRatioBadge');
        if (iRatioBadge) iRatioBadge.innerText = `佔比 ${iRatio}%`;

        let iWeightTotal = 0;
        (sessionData.indivCriteria || []).forEach(crit => iWeightTotal += crit.weight);

        const activeMember = (currentItem && currentItem.members) ? currentItem.members[selectedMemberIndex] : null;
        const memberScores = (activeMember && formState.indivScores) ? (formState.indivScores[activeMember.id] || {}) : {};

        let allIndivEntered = true;
        const indivCriteriaList = sessionData.indivCriteria || [];
        if (indivCriteriaList.length === 0) {
          allIndivEntered = false;
        } else {
          for (let ii = 0; ii < indivCriteriaList.length; ii++) {
            const sc = memberScores[ii];
            if (sc === undefined || sc === null || sc === '') {
              allIndivEntered = false;
              break;
            }
          }
        }

        if (allIndivEntered) {
          let mIndivWeightedSum = 0;
          indivCriteriaList.forEach((crit, ii) => {
            const score = Number(memberScores[ii]) || 0;
            mIndivWeightedSum += score * crit.weight;
          });
          activeMemberIndivSubtotal = iWeightTotal > 0 ? (mIndivWeightedSum / iWeightTotal) : 0;
        } else {
          activeMemberIndivSubtotal = 0;
        }

        if (allGroupEntered && allIndivEntered) {
          combinedFinalScore = totalWeightScale > 0 
            ? ((groupSubtotalScore * gRatio + activeMemberIndivSubtotal * iRatio) / totalWeightScale) 
            : ((groupSubtotalScore + activeMemberIndivSubtotal) / 2);
        } else {
          combinedFinalScore = 0;
        }

        if (groupSubtotalDisplay) groupSubtotalDisplay.innerText = allGroupEntered ? groupSubtotalScore.toFixed(1) : '0.0';
        if (indivSubtotalDisplay) indivSubtotalDisplay.innerText = allIndivEntered ? activeMemberIndivSubtotal.toFixed(1) : '0.0';

        // 當小組分數調整時，自動為全組「已評分同學」重新計算最新加權總分，更新名冊標籤
        if (allGroupEntered && currentItem && currentItem.members) {
          let updatedAny = false;
          currentItem.members.forEach((m, mIdx) => {
            if (mIdx !== selectedMemberIndex && formState.savedMembers && formState.savedMembers[m.id] !== undefined) {
              const mScores = formState.indivScores ? formState.indivScores[m.id] : null;
              if (mScores) {
                let mSum = 0;
                let mWSum = 0;
                (sessionData.indivCriteria || []).forEach((crit, ii) => {
                  const s = mScores[ii];
                  if (s !== undefined && s !== null && s !== '') {
                    mSum += Number(s) * crit.weight;
                    mWSum += crit.weight;
                  }
                });
                if (mWSum > 0) {
                  const mIndivSub = mSum / mWSum;
                  const newRecalculatedScore = totalWeightScale > 0
                    ? parseFloat(((groupSubtotalScore * gRatio + mIndivSub * iRatio) / totalWeightScale).toFixed(1))
                    : 0.0;
                  formState.savedMembers[m.id] = newRecalculatedScore;
                  updatedAny = true;
                }
              }
            }
          });
          if (updatedAny) {
            renderMemberPillTabs();
          }
        }

        if (mainGaugeTitle) mainGaugeTitle.innerText = '加權總分 (團體+個人)';
        const canShowGauge = allGroupEntered && allIndivEntered;
        if (mainGaugeVal) mainGaugeVal.textContent = canShowGauge ? combinedFinalScore.toFixed(1) : '0.0';
        renderSegmentedArc(canShowGauge ? combinedFinalScore : 0);

        if (mainGaugeBreakdown) mainGaugeBreakdown.classList.remove('hidden');
        const gRatioSpan = document.getElementById('gaugeBreakdownGroupRatio');
        if (gRatioSpan) gRatioSpan.innerText = `${gRatio}%`;
        const gScoreSpan = document.getElementById('gaugeBreakdownGroupScore');
        if (gScoreSpan) gScoreSpan.innerText = allGroupEntered ? groupSubtotalScore.toFixed(1) : '0.0';

        const iRatioSpan = document.getElementById('gaugeBreakdownIndivRatio');
        if (iRatioSpan) iRatioSpan.innerText = `${iRatio}%`;
        const iScoreSpan = document.getElementById('gaugeBreakdownIndivScore');
        if (iScoreSpan) iScoreSpan.innerText = allIndivEntered ? activeMemberIndivSubtotal.toFixed(1) : '0.0';

      } else {
        let total = 0, wSum = 0, allEntered = true;
        (sessionData.criteria || []).forEach((crit, i) => {
          const v = formState.scores[i];
          if (v === undefined || v === null || v === '') {
            allEntered = false;
          } else {
            total += Number(v) * crit.weight;
            wSum += crit.weight;
          }
        });
        combinedFinalScore = (allEntered && wSum > 0) ? (total / wSum) : 0;

        if (mainGaugeTitle) mainGaugeTitle.innerText = '個人加權分數';
        if (mainGaugeVal) mainGaugeVal.textContent = allEntered ? combinedFinalScore.toFixed(1) : '0.0';
        renderSegmentedArc(allEntered ? combinedFinalScore : 0);
        if (mainGaugeBreakdown) mainGaugeBreakdown.classList.add('hidden');
      }
    }

    function renderSegmentedArc(score) {
      const group = document.getElementById('segmentedTicksGroup');
      if (!group) return;
      const totalTicks = 26;
      const startAngle = 160;
      const endAngle = 380;
      const angleRange = endAngle - startAngle;
      const cx = 150, cy = 135;
      const rInner = 82, rOuter = 104;

      const numericScore = typeof score === 'number' ? score : parseFloat(score) || 0;
      const activeCount = numericScore > 0 ? Math.round((Math.max(0, Math.min(100, numericScore)) / 100) * totalTicks) : 0;
      let html = '';

      for (let i = 0; i < totalTicks; i++) {
        const ratio = i / (totalTicks - 1);
        const deg = startAngle + ratio * angleRange;
        const rad = (deg * Math.PI) / 180;
        const cos = Math.cos(rad);
        const sin = Math.sin(rad);

        const x1 = (cx + rInner * cos).toFixed(1);
        const y1 = (cy + rInner * sin).toFixed(1);
        const x2 = (cx + rOuter * cos).toFixed(1);
        const y2 = (cy + rOuter * sin).toFixed(1);

        const isActive = i < activeCount && numericScore > 0;
        if (isActive) {
          let strokeColor = '#ca6b2d';
          if (ratio > 0.75) strokeColor = '#fff5ea';
          else if (ratio > 0.45) strokeColor = '#ffcf96';
          else strokeColor = '#ff9f43';

          html += `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" stroke="${strokeColor}" stroke-width="4.2" stroke-linecap="round" filter="url(#segmentGlow)" />`;
        } else {
          html += `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" stroke="#1e2735" stroke-width="3.8" stroke-linecap="round" opacity="0.65" />`;
        }
      }
      group.innerHTML = html;
    }

    function submitCurrentScore() {
      const teacher = document.getElementById('teacherName').value.trim();
      if (!teacher) {
        showToast('請先在頂部輸入「評審老師」姓名！', '⚠️');
        document.getElementById('teacherName').focus();
        return;
      }
      safeStorage.setItem('review_teacher_name', teacher);

      const item = sessionData.items[currentIndex];
      const currentNotes = document.getElementById('notes').value.trim();
      let representativeFinalScore = 0;
      let records = [];
      let syncedCount = 0;

      if (sessionData.isGroupMode) {
        const activeMember = (item.members || [])[selectedMemberIndex] || item.members[0];
        const mScores = (formState.indivScores && formState.indivScores[activeMember.id]) ? formState.indivScores[activeMember.id] : {};

        // 防呆校驗：若小組團體指標尚未填寫完畢，不允許送出
        let allGroupEntered = true;
        (sessionData.groupCriteria || []).forEach((crit, gi) => {
          const s = formState.groupScores && formState.groupScores[gi];
          if (s === undefined || s === null || s === '') {
            allGroupEntered = false;
          }
        });
        if (!allGroupEntered) {
          showToast('請先完成「PART 1 · 小組團體分數」評分！', '⚠️');
          return;
        }

        // 防呆校驗：若個人各項指標尚未填寫完畢，不允許送出
        let allEntered = true;
        (sessionData.indivCriteria || []).forEach((crit, ii) => {
          const s = mScores[ii];
          if (s === undefined || s === null || s === '') {
            allEntered = false;
          }
        });

        if (!allEntered) {
          showToast(`請先完成【${activeMember.name}】的個人評分！`, '⚠️');
          return;
        }

        const gRatio = (sessionData.groupRatio !== undefined && sessionData.groupRatio > 0) ? sessionData.groupRatio : 30;
        const iRatio = (sessionData.indivRatio !== undefined && sessionData.indivRatio > 0) ? sessionData.indivRatio : 40;
        const totalWeightScale = gRatio + iRatio;

        let gWeightedSum = 0;
        let gWeightTotal = 0;
        (sessionData.groupCriteria || []).forEach((crit, gi) => {
          const s = Number(formState.groupScores[gi]) || 0;
          gWeightedSum += s * crit.weight;
          gWeightTotal += crit.weight;
        });
        const groupSubtotalScore = gWeightTotal > 0 ? (gWeightedSum / gWeightTotal) : 0;

        let iWeightTotal = 0;
        (sessionData.indivCriteria || []).forEach(crit => iWeightTotal += crit.weight);

        let mIndivWeightedSum = 0;
        (sessionData.indivCriteria || []).forEach((crit, ii) => {
          const s = Number(mScores[ii]) || 0;
          mIndivWeightedSum += s * crit.weight;
        });
        const mIndivSubtotal = iWeightTotal > 0 ? (mIndivWeightedSum / iWeightTotal) : 0;

        const finalScoreVal = totalWeightScale > 0 
          ? parseFloat(((groupSubtotalScore * gRatio + mIndivSubtotal * iRatio) / totalWeightScale).toFixed(1))
          : 0.0;

        representativeFinalScore = finalScoreVal;

        // 當前被選中當事者
        records = [{
          id: activeMember.id,
          name: activeMember.name,
          advisor: activeMember.advisor,
          finalScore: finalScoreVal,
          notes: currentNotes
        }];

        // 自動將本組其他「已評過分」的同學一起打包送出更新（確保連動最新團體成績到試算表）
        (item.members || []).forEach(m => {
          if (m.id !== activeMember.id && formState.savedMembers && formState.savedMembers[m.id] !== undefined) {
            records.push({
              id: m.id,
              name: m.name,
              advisor: m.advisor,
              finalScore: formState.savedMembers[m.id],
              notes: (formState.memberNotes && formState.memberNotes[m.id]) ? formState.memberNotes[m.id] : ''
            });
            syncedCount++;
          }
        });

        formState.memberNotes[activeMember.id] = currentNotes;
        if (!formState.savedMembers) formState.savedMembers = {};
        formState.savedMembers[activeMember.id] = finalScoreVal;

        if (!reviewMemory[currentIndex]) reviewMemory[currentIndex] = {};
        reviewMemory[currentIndex].groupScores = { ...formState.groupScores };
        reviewMemory[currentIndex].indivScores = JSON.parse(JSON.stringify(formState.indivScores));
        reviewMemory[currentIndex].memberNotes = { ...formState.memberNotes };
        reviewMemory[currentIndex].savedMembers = { ...formState.savedMembers };
        reviewMemory[currentIndex].representativeScore = representativeFinalScore;
        reviewMemory[currentIndex].notes = currentNotes;
        reviewMemory[currentIndex].photo = formState.currentPhoto;

      } else {
        let total = 0, wSum = 0, allEntered = true;
        (sessionData.criteria || []).forEach((crit, i) => {
          const v = formState.scores[i];
          if (v === undefined || v === null || v === '') {
            allEntered = false;
          } else {
            total += Number(v) * crit.weight;
            wSum += crit.weight;
          }
        });

        if (!allEntered) {
          showToast(`請先完成【${item.name}】的各項評分！`, '⚠️');
          return;
        }

        const finalVal = wSum > 0 ? (total / wSum) : 0;
        representativeFinalScore = parseFloat(finalVal.toFixed(1));

        records = [{
          id: item.id,
          name: item.name,
          advisor: item.advisor,
          finalScore: representativeFinalScore
        }];

        reviewMemory[currentIndex] = {
          scores: { ...formState.scores },
          representativeScore: representativeFinalScore,
          notes: currentNotes,
          photo: formState.currentPhoto
        };
      }

      setButtonLoading(true);

      const safetyTimer = setTimeout(() => {
        setButtonLoading(false);
      }, 6000);

      const payload = {
        grade: sessionData.grade,
        teacher: teacher,
        groupName: sessionData.isGroupMode ? item.groupName : '個人專題',
        notes: currentNotes,
        imageBase64: formState.currentPhoto,
        records: records
      };

      backendService.saveScores(
        payload,
        res => {
          clearTimeout(safetyTimer);

          if (res && res.success) {
            if (sessionData.isGroupMode) {
              const currentItem = sessionData.items[currentIndex];
              const totalMembers = (currentItem.members || []).length;
              const activeMember = currentItem.members[selectedMemberIndex];
              const currentMemberName = activeMember ? activeMember.name : '';

              if (syncedCount > 0) {
                showToast(`已儲存【${currentMemberName}】(${representativeFinalScore}分)，並同步更新全組 ${syncedCount} 位同學最新成績！`, '✔');
              } else {
                showToast(`已成功儲存【${currentMemberName}】(${representativeFinalScore} 分)！`, '✔');
              }

              if (selectedMemberIndex + 1 < totalMembers) {
                selectMember(selectedMemberIndex + 1);
                window.scrollTo({ top: 80, behavior: 'smooth' });
              } else {
                setButtonLoading(false);
                showToast(`🎉 第 ${currentItem.groupName} 組已全員完成評分！`, '🏆');
                if (currentIndex + 1 < sessionData.items.length) {
                  renderView(currentIndex + 1);
                } else {
                  renderView(currentIndex);
                }
              }
            } else {
              setButtonLoading(false);
              showToast(res.isMock ? '評分已儲存！' : '已成功儲存至 Google 試算表！', '✔');
              if (currentIndex + 1 < sessionData.items.length) {
                renderView(currentIndex + 1);
              } else {
                showToast('🎉 已完成全體評分！', '🏆');
                renderView(currentIndex);
              }
            }
          } else {
            setButtonLoading(false);
            showToast('儲存失敗：' + (res ? res.error : '未知錯誤'), '❌');
          }
        },
        err => {
          clearTimeout(safetyTimer);
          setButtonLoading(false);
          showToast('網路或伺服器錯誤：' + (err.message || err), '⚠️');
        }
      );
    }

    function handleImageCaptured(e) {
      const file = e.target.files[0];
      if (!file) return;

      const photoContainer = document.getElementById('photoContainer');
      if (photoContainer) {
        photoContainer.innerHTML = `
          <div class="py-3 bg-[#10141a] border border-[#ca6b2d]/50 rounded-xl flex items-center justify-center gap-2 text-[#ff9f43] text-xs font-times animate-pulse">
            <span class="animate-spin">⏳</span> Compressing Image...
          </div>
        `;
      }

      const reader = new FileReader();
      reader.onload = function(event) {
        const img = new Image();
        img.onload = function() {
          const canvas = document.createElement('canvas');
          const maxDim = 600;
          let width = img.width;
          let height = img.height;

          if (width > height && width > maxDim) {
            height = Math.round((height * maxDim) / width);
            width = maxDim;
          } else if (height > maxDim) {
            width = Math.round((width * maxDim) / height);
            height = maxDim;
          }

          canvas.width = width;
          canvas.height = height;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, width, height);

          const compressedBase64 = canvas.toDataURL('image/jpeg', 0.5);
          formState.currentPhoto = compressedBase64;
          
          if (photoContainer) {
            photoContainer.innerHTML = renderPhotoSectionHTML();
          }
        };
        img.src = event.target.result;
      };
      reader.readAsDataURL(file);
      e.target.value = '';
    }

    function updateFilterTabButtons() {
      const tabs = ['all', 'done', 'pending'];
      tabs.forEach(t => {
        const btn = document.getElementById('tabFilter' + t.charAt(0).toUpperCase() + t.slice(1));
        if (btn) {
          if (t === currentOverviewFilter) {
            btn.className = 'px-3 py-1.5 rounded-xl metallic-bronze-btn text-white font-extrabold transition shadow-sm font-times';
          } else {
            btn.className = 'px-3 py-1.5 rounded-xl bg-[#161c24] text-[#8e9aa8] hover:bg-[#202733] transition border border-[#2f3947] font-times';
          }
        }
      });
    }

      if (query) {
        clearBtn.classList.remove('hidden');
      } else {

    function renderOverviewCards() {
      if (!sessionData || !sessionData.items) return;

      const container = document.getElementById('overviewCardsList');
      const query = document.getElementById('modalSearchInput').value.trim().toLowerCase();
      
      let totalCount = sessionData.items.length;
      let doneCount = 0;
      let pendingCount = 0;

      sessionData.items.forEach((_, idx) => {
        if (reviewMemory[idx]) doneCount++;
        else pendingCount++;
      });

      document.getElementById('countAll').innerText = totalCount;
      document.getElementById('countDone').innerText = doneCount;
      document.getElementById('countPending').innerText = pendingCount;
      document.getElementById('modalProgressBadge').innerText = `${doneCount}/${totalCount} Completed`;

      let filteredItems = [];

      sessionData.items.forEach((item, idx) => {
        const isDone = !!reviewMemory[idx];

        if (currentOverviewFilter === 'done' && !isDone) return;
        if (currentOverviewFilter === 'pending' && isDone) return;

        if (query) {
          let match = false;
          if (sessionData.isGroupMode) {
            if (item.groupName && item.groupName.toLowerCase().includes(query)) match = true;
            if (item.advisor && item.advisor.toLowerCase().includes(query)) match = true;
            if (item.members) {
              item.members.forEach(m => {
                if ((m.name && m.name.toLowerCase().includes(query)) || (m.id && m.id.toLowerCase().includes(query))) match = true;
              });
            }
          } else {
            if ((item.name && item.name.toLowerCase().includes(query)) || (item.id && item.id.toLowerCase().includes(query)) || (item.advisor && item.advisor.toLowerCase().includes(query))) {
              match = true;
            }
          }
          if (!match) return;
        }

        filteredItems.push({ item: item, originalIndex: idx, isDone: isDone });
      });

      if (filteredItems.length === 0) {
        container.innerHTML = `
          <div class="py-12 text-center text-[#8e9aa8] font-medium text-xs font-times italic">
            No matching records found
          </div>
        `;
        return;
      }

      container.innerHTML = filteredItems.map(({ item, originalIndex, isDone }) => {
        const isCurrent = originalIndex === currentIndex;
        const mem = reviewMemory[originalIndex];
        const photo = mem && mem.photo ? mem.photo : null;
        const scoreDisplay = mem && mem.representativeScore ? `${mem.representativeScore.toFixed(1)} PTS` : (isDone ? '已評分' : '待評分');

        let titleText = sessionData.isGroupMode ? `第 ${item.groupName} 組` : item.name;
        let subtitleText = sessionData.isGroupMode && item.members ? item.members.map(m => m.name).join(', ') : `#${item.id}`;
        let advisorText = item.advisor || '未指定';

        return `
          <div onclick="jumpToItem(${originalIndex})" class="p-3 rounded-xl border ${isCurrent ? 'border-[#ca6b2d] bg-[#19202a] shadow-[0_0_15px_rgba(202,107,45,0.35)]' : 'border-[#212833] bg-[#11151c] hover:border-[#354050]'} active:scale-[0.98] transition cursor-pointer flex items-center justify-between gap-3 shadow-md">
            <div class="flex items-center gap-3 min-w-0">
              ${photo ? `
                <img src="${photo}" class="w-11 h-11 object-cover rounded-lg border border-[#ca6b2d]/70 shadow-sm shrink-0">
              ` : `
                <div class="w-11 h-11 rounded-lg bg-[#161c24] border border-[#2e3745] flex items-center justify-center font-times font-bold text-sm text-[#ff9f43] shrink-0">
                  ${originalIndex + 1}
                </div>
              `}
              <div class="min-w-0">
                <div class="flex items-center gap-2">
                  <span class="text-sm font-black text-white truncate">${titleText}</span>
                  ${isCurrent ? `<span class="text-[9px] font-times font-bold bg-[#ca6b2d] text-white px-1.5 py-0.5 rounded shadow-sm">NOW</span>` : ''}
                </div>
                <div class="text-[11px] text-[#8e9aa8] font-times truncate">${subtitleText} ｜ Advisor: ${advisorText}</div>
              </div>
            </div>

            <div class="text-right shrink-0">
              ${isDone ? `
                <div class="bg-[#1b212a] border border-[#ca6b2d]/60 text-[#ffb573] font-times font-bold text-xs px-2.5 py-1 rounded-lg shadow-inner">
                  ${scoreDisplay}
                </div>
              ` : `
                <div class="bg-[#141820] text-[#616e7e] font-times font-bold text-xs px-2.5 py-1 rounded-lg border border-[#212733]">
                  PENDING
                </div>
              `}
            </div>
          </div>
        `;
      }).join('');
    }
